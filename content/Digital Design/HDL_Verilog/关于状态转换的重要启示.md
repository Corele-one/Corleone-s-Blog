在编写FSM时，==一个状态只做一件事情==，比如说下面这个状态机，如果在接收了8位数据之后，还没有接收到停止位的话，那么就不会有DONE信号发出来。而我之前的代码，是在REC状态下处理这件事情，在REC状态下比较计数器和停止位，会导致状态转移混乱，以及可能的时序错误。
https://www.doubao.com/thread/w90edf424bab9ed54
```verilog
module top_module(
    input clk,
    input in,
    input reset,    // Synchronous reset
    output done
); 
    reg [2:0] state;
    reg [2:0] next_state;
    parameter IDLE	=3'b000;
    parameter REC 	=3'b001;
    parameter DONE	=3'b010;
    parameter ERROR	=3'b011;
    parameter CHECK =3'b100;
    //state transision
    reg[3:0] cnt;
    always@(*) begin
        case(state)
            IDLE:begin
                if(in) begin
                    next_state=IDLE; 
                end
                else begin
                    next_state=REC;
                end
            end
            REC:begin
                if(cnt==4'd7) begin
                    next_state=CHECK;
                end
                else begin
                    next_state=REC;
                end
            end
            CHECK:begin
                if(in) begin
                    next_state=DONE;
                end
                else begin
                    next_state=ERROR;
                end
            end
            DONE:begin
                if(in) begin
                	next_state=IDLE;
                end
                else begin
                    next_state=REC;
                end
            end
            ERROR:begin
                if(in) begin
                   next_state=IDLE; 
                end
                else begin
                   next_state=ERROR; 
                end
            end
            default:next_state=IDLE;
        endcase
    end
    //next_state
    always@(posedge clk) begin
        if(reset)  begin
            state<=IDLE;
        end
        else begin
            state<=next_state;
        end 
    end
    //cnt
    
    always@(posedge clk) begin
        if(reset|(state!=REC)) begin
            cnt<=0;
        end 
        else begin
            cnt<=cnt+1'b1;
        end
    end
    assign done=(state==DONE);
endmodule
```
